import pathlib
import base64

apikey = "fly-***"
model = "flymyai/bytedance-seedream-5_0_pro"
payload = {
    "image": None,
    "image1": None,
    "image2": None,
    "image3": None,
    "image4": None,
    "image5": None,
    "image6": None,
    "image7": None,
    "image8": None,
    "image9": None,
    "image10": None,
    "image11": None,
    "image12": None,
    "image13": None,
    "size": "2k",
    "watermark": False,
    "sequential_image_generation": "disabled",
    "optimize_prompt_mode": "standard"
}

fma_client = client(apikey=apikey)

try:
    response = fma_client.predict(model=model, payload=payload)
    image_encoded = response.output_data["image"][0]
    image = base64.b64decode(image_encoded)
    output_path = pathlib.Path("image_output.jpg")
    output_path.write_bytes(image)
    print(f"Saved {output_path.name}")
except FlyMyAIPredictException as error:
    print(error)
